### 注册流程
1. 验证邀请码有效性
2. 检查邮箱/手机号是否已注册
3. 密码加密（bcrypt）
4. 创建 User 记录
5. 创建 Tenant 记录（关联 User）
6. 更新邀请关系（invitedBy）
7. 生成 JWT Token
8. 返回 Token 和用户信息

import { IsEmail, IsString, MinLength, IsNotEmpty, IsOptional } from 'class-validator';

export class RegisterDto {
  @IsEmail({}, { message: '邮箱格式不正确' })
  @IsNotEmpty({ message: '邮箱不能为空' })
  email: string;

  @IsString()
  @MinLength(6, { message: '密码至少6位' })
  password: string;

  @IsString()
  @IsNotEmpty({ message: '邀请码不能为空' })
  inviteCode: string;

  @IsOptional()
  @IsString()
  phone?: string;
}

export class RegisterResponseDto {
  accessToken: string;
  refreshToken: string;
  user: {
    id: string;
    email: string;
    role: string;
  };
}

### 步骤2：创建 AuthController

**backend/src/modules/auth/auth.controller.ts**

```typescript
import { Controller, Post, Body, BadRequestException, ConflictException } from '@nestjs/common';
import { RegisterDto, RegisterResponseDto } from './dto/register.dto';
import { InviteCodeService } from './invite-code.service';
import { JwtAuthService } from './jwt.service';
import { PrismaService } from '../prisma/prisma.service';
import * as bcrypt from 'bcrypt';

@Controller('auth')
export class AuthController {
  constructor(
    private readonly inviteCodeService: InviteCodeService,
    private readonly jwtService: JwtAuthService,
    private readonly prisma: PrismaService,
  ) {}

  @Post('register')
  async register(@Body() dto: RegisterDto): Promise<RegisterResponseDto> {
    // 1. 验证邀请码
    const validation = await this.inviteCodeService.validateInviteCode(dto.inviteCode);
    if (!validation.valid) {
      throw new BadRequestException('邀请码无效');
    }

    // 2. 检查邮箱是否已注册
    const existingUser = await this.prisma.user.findUnique({
      where: { email: dto.email },
    });
    if (existingUser) {
      throw new ConflictException('邮箱已被注册');
    }

    // 3. 检查手机号是否已注册（如果提供）
    if (dto.phone) {
      const existingPhone = await this.prisma.user.findUnique({
        where: { phone: dto.phone },
      });
      if (existingPhone) {
        throw new ConflictException('手机号已被注册');
      }
    }

    // 4. 密码加密
    const passwordHash = await bcrypt.hash(dto.password, 10);

    // 5. 创建用户和租客（事务）
    const user = await this.prisma.$transaction(async (tx) => {
      // 创建 User
      const newUser = await tx.user.create({
        data: {
          email: dto.email,
          phone: dto.phone,
          passwordHash,
          role: 'TENANT',
          invitedBy: validation.invitedBy,
          isActive: true,
        },
      });

      // 创建 Tenant
      await tx.tenant.create({
        data: {
          userId: newUser.id,
          buildingId: process.env.DEFAULT_BUILDING_ID || 'building-001',
          status: 'PENDING',
          emergencyContact: dto.phone || '',
          emergencyPhone: dto.phone || '',
        },
      });

      return newUser;
    });

    // 6. 生成 JWT Token
    const tokens = await this.jwtService.generateTokens({
      sub: user.id,
      role: user.role,
      buildingId: process.env.DEFAULT_BUILDING_ID,
    });

    // 7. 返回结果
    return {
      ...tokens,
      user: {
        id: user.id,
        email: user.email,
        role: user.role,
      },
    };
  }
}
```

### 步骤3：更新 AuthModule

**backend/src/modules/auth/auth.module.ts**

```typescript
import { Module } from '@nestjs/common';
import { ConfigModule, ConfigService } from '@nestjs/config';
import { JwtModule } from '@nestjs/jwt';
import { PassportModule } from '@nestjs/passport';
import { PrismaModule } from '../prisma/prisma.module.js';
import { JwtAuthService } from './jwt.service.js';
import { JwtStrategy } from './strategies/jwt.strategy.js';
import { InviteCodeService } from './invite-code.service.js';
import { AuthController } from './auth.controller.js';
import jwtConfig from '../../config/jwt.config.js';

@Module({
  imports: [
    ConfigModule.forFeature(jwtConfig),
    PassportModule.register({ defaultStrategy: 'jwt' }),
    JwtModule.registerAsync({
      imports: [ConfigModule],
      useFactory: (configService: ConfigService) => ({
        secret: configService.get('jwt.secret'),
        signOptions: {
          expiresIn: configService.get('jwt.accessTokenExpiry'),
        },
      }),
      inject: [ConfigService],
    }),
    PrismaModule,
  ],
  controllers: [AuthController],
  providers: [JwtAuthService, JwtStrategy, InviteCodeService],
  exports: [JwtAuthService, InviteCodeService, PassportModule],
})
export class AuthModule {}
```

## 安全考虑

1. **密码加密**：使用 bcrypt，salt rounds = 10
2. **邀请码验证**：必须提供有效邀请码
3. **唯一性检查**：邮箱和手机号不能重复
4. **事务保证**：User 和 Tenant 创建在同一事务中
5. **敏感信息**：返回结果不包含密码哈希
