-- 1. 非月付合同的租金账单，对照「月租金 x 周期月数」。
--    首末期不足月不在这条里下结论。
SELECT
  c.id AS contract_id,
  c."contractNumber",
  c."paymentCycle",
  c."rentAmount",
  b.id AS bill_id,
  b.period,
  b.amount,
  b."paidAmount",
  b.status
FROM contracts c
JOIN bills b ON b."contractId" = c.id
WHERE b."billType" = 'RENT'
  AND b.status <> 'CANCELLED'
  AND c."paymentCycle" <> 'MONTHLY'
  AND b.amount IS DISTINCT FROM CASE c."paymentCycle"
    WHEN 'QUARTERLY' THEN c."rentAmount" * 3
    WHEN 'SEMI_ANNUAL' THEN c."rentAmount" * 6
    WHEN 'ANNUAL' THEN c."rentAmount" * 12
    ELSE c."rentAmount"
  END
ORDER BY c.id, b.period;

-- 2. 账单已付本金与有效收款本金分配不一致。
SELECT
  b.id AS bill_id,
  b."billNumber",
  b.amount,
  b."paidAmount",
  b.status,
  COALESCE(SUM(p."billPortion"), 0) AS confirmed_bill_portion
FROM bills b
LEFT JOIN payment_records p
  ON p."billId" = b.id
 AND p.status = 'CONFIRMED'
 AND p."voidedAt" IS NULL
 AND p."reversalOfId" IS NULL
GROUP BY b.id, b."billNumber", b.amount, b."paidAmount", b.status
HAVING b."paidAmount" IS DISTINCT FROM COALESCE(SUM(p."billPortion"), 0)
ORDER BY b.id;

-- 3. 押金余额与有符号流水合计不一致。
SELECT
  l.id AS ledger_id,
  l."contractId",
  l.balance,
  COALESCE(SUM(t.amount), 0) AS signed_sum
FROM deposit_ledgers l
LEFT JOIN deposit_transactions t ON t."ledgerId" = l.id
GROUP BY l.id, l."contractId", l.balance
HAVING l.balance IS DISTINCT FROM COALESCE(SUM(t.amount), 0)
ORDER BY l.id;

-- 4. 退租房间与合同房间不一致。
SELECT r.id, r."contractId", r."roomId" AS checkout_room_id, c."roomId" AS contract_room_id
FROM checkout_records r
JOIN contracts c ON c.id = r."contractId"
WHERE r."roomId" IS DISTINCT FROM c."roomId";

-- 5. 已完成退租缺少明细，或没有押金台账。缺台账不等于零押金。
SELECT r.id, r."contractId", r.status
FROM checkout_records r
WHERE r.status = 'COMPLETED'
  AND (
    NOT EXISTS (SELECT 1 FROM checkout_items i WHERE i."checkoutId" = r.id)
    OR NOT EXISTS (SELECT 1 FROM deposit_ledgers l WHERE l."contractId" = r."contractId")
  );

-- 6. 同一合同、同一租金账期多笔未取消账单。
SELECT b."contractId", b.period, COUNT(*) AS bill_count
FROM bills b
WHERE b."billType" = 'RENT' AND b.status <> 'CANCELLED'
GROUP BY b."contractId", b.period
HAVING COUNT(*) > 1;

-- 7. 已过结束日但仍 ACTIVE 的合同。只列出，不自动终止。
SELECT c.id, c."contractNumber", c."endDate", c."roomId"
FROM contracts c
WHERE c.status = 'ACTIVE' AND c."endDate" < CURRENT_DATE
ORDER BY c."endDate", c.id;
