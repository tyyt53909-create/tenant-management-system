## 任务目标

实现邀请码的生成、验证、失效逻辑，确保只有持有有效邀请码的用户才能注册账号。

---

## 任务范围

### 必须完成
1. InviteCodeService（邀请码业务逻辑）
2. 生成邀请码（管理员专用）
3. 验证邀请码（注册时使用）
4. 邀请码失效逻辑（已使用/已过期）
5. DTO 定义（GenerateInviteCodeDto, ValidateInviteCodeDto）
6. 单元测试

### 邀请码规则
- 格式：`INVITE-{timestamp}` 或自定义
- 一次性使用（使用后失效）
- 有效期：默认365天
- 管理员可批量生成
- 租客注册时验证并绑定

## 详细实现

### 步骤1：创建 InviteCodeService

**backend/src/modules/auth/invite-code.service.ts**

```typescript
import { Injectable, BadRequestException, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class InviteCodeService {
  constructor(private readonly prisma: PrismaService) {}

  /**
   * 生成邀请码（管理员专用）
   */
  async generateInviteCode(
    adminUserId: string,
    expiryDays: number = 365,
  ): Promise<string> {
    const inviteCode = `INVITE-${Date.now()}-${Math.random().toString(36).substring(2, 8).toUpperCase()}`;
    const expiredAt = new Date(Date.now() + expiryDays * 24 * 60 * 60 * 1000);

    await this.prisma.user.update({
      where: { id: adminUserId },
      data: {
        inviteCode,
        inviteExpiredAt: expiredAt,
      },
    });

    return inviteCode;
  }

  /**
   * 验证邀请码（注册时使用）
   */
  async validateInviteCode(code: string): Promise<{ valid: boolean; userId?: string }> {
    const user = await this.prisma.user.findFirst({
      where: {
        inviteCode: code,
      },
      select: {
        id: true,
        inviteCode: true,
        inviteExpiredAt: true,
        inviteUsedAt: true,
      },
    });

    // 邀请码不存在
    if (!user) {
      return { valid: false };
    }

    // 邀请码已使用
    if (user.inviteUsedAt) {
      throw new BadRequestException('邀请码已被使用');
    }

    // 邀请码已过期
    if (user.inviteExpiredAt && new Date() > user.inviteExpiredAt) {
      throw new BadRequestException('邀请码已过期');
    }

    return { valid: true, userId: user.id };
  }

  /**
   * 标记邀请码为已使用
   */
  async markInviteCodeAsUsed(code: string, usedByUserId: string): Promise<void> {
    await this.prisma.user.updateMany({
      where: { inviteCode: code },
      data: {
        inviteUsedAt: new Date(),
        inviteUsedBy: usedByUserId,
      },
    });
  }

  /**
   * 检查用户是否有有效的邀请码
   */
  async hasValidInviteCode(userId: string): Promise<boolean> {
    const user = await this.prisma.user.findUnique({
      where: { id: userId },
      select: {
        inviteCode: true,
        inviteExpiredAt: true,
        inviteUsedAt: true,
      },
    });

    if (!user || !user.inviteCode) {
      return false;
    }

    // 已使用
    if (user.inviteUsedAt) {
      return false;
    }

    // 已过期
    if (user.inviteExpiredAt && new Date() > user.inviteExpiredAt) {
      return false;
    }

    return true;
  }
}
```

import { Module } from '@nestjs/common';
import { ConfigModule, ConfigService } from '@nestjs/config';
import { JwtModule } from '@nestjs/jwt';
import { PassportModule } from '@nestjs/passport';
import { PrismaModule } from '../prisma/prisma.module.js';
import { JwtAuthService } from './jwt.service.js';
import { JwtStrategy } from './strategies/jwt.strategy.js';
import { InviteCodeService } from './invite-code.service.js';
import jwtConfig from '../../config/jwt.config.js';

@Module({
  imports: [
    ConfigModule.forFeature(jwtConfig),
    PassportModule.register({ defaultStrategy: 'jwt' }),
    JwtModule.registerAsync({
      imports: [ConfigModule],
      useFactory: (configService: ConfigService) => ({
        secret: configService.get('jwt.secret'),
        signOptions: {
          expiresIn: configService.get('jwt.accessTokenExpiry'),
        },
      }),
      inject: [ConfigService],
    }),
    PrismaModule,
  ],
  providers: [JwtAuthService, JwtStrategy, InviteCodeService],
  exports: [JwtAuthService, InviteCodeService, PassportModule],
})
export class AuthModule {}

import { Test, TestingModule } from '@nestjs/testing';
import { BadRequestException } from '@nestjs/common';
import { InviteCodeService } from './invite-code.service';
import { PrismaService } from '../prisma/prisma.service';

describe('InviteCodeService', () => {
  let service: InviteCodeService;
  let prisma: PrismaService;

  beforeEach(async () => {
    const module: TestingModule = await Test.createTestingModule({
      providers: [
        InviteCodeService,
        {
          provide: PrismaService,
          useValue: {
            user: {
              update: vi.fn(),
              findFirst: vi.fn(),
              findUnique: vi.fn(),
              updateMany: vi.fn(),
            },
          },
        },
      ],
    }).compile();

    service = module.get<InviteCodeService>(InviteCodeService);
    prisma = module.get<PrismaService>(PrismaService);
  });

  it('应该定义服务', () => {
    expect(service).toBeDefined();
  });

  describe('generateInviteCode', () => {
    it('应该生成邀请码', async () => {
      const userId = 'admin-123';
      const expiryDays = 365;

      vi.spyOn(prisma.user, 'update').mockResolvedValue({} as any);

      const inviteCode = await service.generateInviteCode(userId, expiryDays);

      expect(inviteCode).toMatch(/^INVITE-\d+-[A-Z0-9]{6}$/);
      expect(prisma.user.update).toHaveBeenCalledWith({
        where: { id: userId },
        data: {
          inviteCode: expect.any(String),
          inviteExpiredAt: expect.any(Date),
        },
      });
    });
  });

  describe('validateInviteCode', () => {
    it('应该验证有效的邀请码', async () => {
      const code = 'INVITE-123-ABC123';
      const mockUser = {
        id: 'user-123',
        inviteCode: code,
        inviteExpiredAt: new Date(Date.now() + 365 * 24 * 60 * 60 * 1000),
        inviteUsedAt: null,
      };

      vi.spyOn(prisma.user, 'findFirst').mockResolvedValue(mockUser as any);

      const result = await service.validateInviteCode(code);

      expect(result.valid).toBe(true);
      expect(result.userId).toBe('user-123');
    });

    it('应该拒绝不存在的邀请码', async () => {
      vi.spyOn(prisma.user, 'findFirst').mockResolvedValue(null);

      const result = await service.validateInviteCode('INVALID');

      expect(result.valid).toBe(false);
    });

    it('应该拒绝已使用的邀请码', async () => {
      const mockUser = {
        id: 'user-123',
        inviteCode: 'INVITE-123',
        inviteExpiredAt: new Date(Date.now() + 365 * 24 * 60 * 60 * 1000),
        inviteUsedAt: new Date(),
      };

      vi.spyOn(prisma.user, 'findFirst').mockResolvedValue(mockUser as any);

      await expect(service.validateInviteCode('INVITE-123')).rejects.toThrow(
        BadRequestException,
      );
    });

    it('应该拒绝已过期的邀请码', async () => {
      const mockUser = {
        id: 'user-123',
        inviteCode: 'INVITE-123',
        inviteExpiredAt: new Date(Date.now() - 1000), // 已过期
        inviteUsedAt: null,
      };

      vi.spyOn(prisma.user, 'findFirst').mockResolvedValue(mockUser as any);

      await expect(service.validateInviteCode('INVITE-123')).rejects.toThrow(
        BadRequestException,
      );
    });
  });

  describe('markInviteCodeAsUsed', () => {
    it('应该标记邀请码为已使用', async () => {
      const code = 'INVITE-123';
      const usedByUserId = 'new-user-123';

      vi.spyOn(prisma.user, 'updateMany').mockResolvedValue({ count: 1 } as any);

      await service.markInviteCodeAsUsed(code, usedByUserId);

      expect(prisma.user.updateMany).toHaveBeenCalledWith({
        where: { inviteCode: code },
        data: {
          inviteUsedAt: expect.any(Date),
          inviteUsedBy: usedByUserId,
        },
      });
    });
  });
});
