```typescript
// rooms/rooms.service.spec.ts
describe('RoomsService', () => {
  let service: RoomsService;
  let prisma: DeepMockProxy<PrismaClient>;

  beforeEach(() => {
    prisma = mockDeep<PrismaClient>();
    service = new RoomsService(prisma);
  });

  describe('isAvailableForRent', () => {
    it('应返回 true 当房间状态为 VACANT 且已清洁', () => {
      const room = {
        status: RoomStatus.VACANT,
        cleanStatus: CleanStatus.CLEAN,
      };
      
      expect(service.isAvailableForRent(room)).toBe(true);
    });

    it('应返回 false 当房间状态为 VACANT 但未清洁', () => {
      const room = {
        status: RoomStatus.VACANT,
        cleanStatus: CleanStatus.PENDING_CLEAN,
      };
      
      expect(service.isAvailableForRent(room)).toBe(false);
    });

    it('应返回 false 当房间状态为 OCCUPIED', () => {
      const room = {
        status: RoomStatus.OCCUPIED,
        cleanStatus: CleanStatus.CLEAN,
      };
      
      expect(service.isAvailableForRent(room)).toBe(false);
    });
  });

  describe('markAsClean', () => {
    it('应成功标记房间为已清洁', async () => {
      const roomId = 'room-123';
      const mockRoom = {
        id: roomId,
        status: RoomStatus.VACANT,
        cleanStatus: CleanStatus.PENDING_CLEAN,
        version: 1,
      };

      prisma.room.findUnique.mockResolvedValue(mockRoom);
      prisma.room.update.mockResolvedValue({
        ...mockRoom,
        cleanStatus: CleanStatus.CLEAN,
        version: 2,
      });

      const result = await service.markAsClean(roomId, 'user-123');

      expect(result.cleanStatus).toBe(CleanStatus.CLEAN);
      expect(prisma.room.update).toHaveBeenCalledWith({
        where: { id: roomId, version: 1 },
        data: {
          cleanStatus: CleanStatus.CLEAN,
          version: { increment: 1 },
        },
      });
    });

    it('应抛出异常当房间不存在', async () => {
      prisma.room.findUnique.mockResolvedValue(null);

      await expect(
        service.markAsClean('non-existent', 'user-123')
      ).rejects.toThrow('房间不存在');
    });

    it('应抛出异常当房间状态不是 VACANT', async () => {
      const mockRoom = {
        id: 'room-123',
        status: RoomStatus.OCCUPIED,
        cleanStatus: CleanStatus.PENDING_CLEAN,
      };

      prisma.room.findUnique.mockResolvedValue(mockRoom);

      await expect(
        service.markAsClean('room-123', 'user-123')
      ).rejects.toThrow('只能清洁空置房间');
    });
  });
});
```

#### 工具函数测试（计算逻辑）

```typescript
// billing/utils/late-fee.util.spec.ts
describe('LateFeeUtil', () => {
  describe('calculateLateFee', () => {
    it('应正确计算滞纳金（日比例方式）', () => {
      const rule = {
        graceDays: 3,
        dailyRate: 0.003,  // 千分之三
        cap: 50000,        // 封顶500元
      };
      const baseAmount = 200000;  // 2000元
      const overdueDays = 10;

      // 前3天免费，后7天收费
      // 2000 * 0.003 * 7 = 42元
      const result = LateFeeUtil.calculate(rule, baseAmount, overdueDays);

      expect(result.amount).toBe(4200);  // 42元 = 4200分
      expect(result.effectiveDays).toBe(7);
      expect(result.capped).toBe(false);
    });

    it('应正确处理封顶逻辑', () => {
      const rule = {
        graceDays: 3,
        dailyRate: 0.01,   // 百分之一
        cap: 10000,        // 封顶100元
      };
      const baseAmount = 200000;
      const overdueDays = 20;

      // 2000 * 0.01 * 17 = 340元 > 100元封顶
      const result = LateFeeUtil.calculate(rule, baseAmount, overdueDays);

      expect(result.amount).toBe(10000);  // 封顶100元
      expect(result.capped).toBe(true);
    });

    it('应返回 0 当逾期天数在宽限期内', () => {
      const rule = {
        graceDays: 3,
        dailyRate: 0.003,
        cap: 50000,
      };
      const baseAmount = 200000;
      const overdueDays = 2;

      const result = LateFeeUtil.calculate(rule, baseAmount, overdueDays);

      expect(result.amount).toBe(0);
      expect(result.effectiveDays).toBe(0);
    });
  });
});
```

```typescript
// rooms/rooms.e2e.spec.ts
describe('Rooms API (e2e)', () => {
  let app: INestApplication;
  let prisma: PrismaService;
  let adminToken: string;
  let tenantToken: string;

  beforeAll(async () => {
    const moduleFixture = await Test.createTestingModule({
      imports: [AppModule],
    }).compile();

    app = moduleFixture.createNestApplication();
    prisma = app.get(PrismaService);
    
    await app.init();
    
    // 创建测试用户并获取 Token
    adminToken = await createTestAdmin(prisma);
    tenantToken = await createTestTenant(prisma);
  });

  afterAll(async () => {
    await prisma.$disconnect();
    await app.close();
  });

  describe('GET /rooms', () => {
    it('应返回房间列表（管理者）', async () => {
      const response = await request(app.getHttpServer())
        .get('/api/v1/rooms')
        .set('Authorization', `Bearer ${adminToken}`)
        .expect(200);

      expect(response.body.success).toBe(true);
      expect(response.body.data).toBeInstanceOf(Array);
      expect(response.body.meta.pagination).toBeDefined();
    });

    it('应返回待清洁房间列表（保洁）', async () => {
      // 创建保洁用户
      const cleanerToken = await createTestCleaner(prisma);
      
      const response = await request(app.getHttpServer())
        .get('/api/v1/rooms')
        .set('Authorization', `Bearer ${cleanerToken}`)
        .expect(200);

      // 保洁只能看到待清洁房间
      expect(response.body.data.every(room => 
        room.cleanStatus === 'PENDING_CLEAN'
      )).toBe(true);
    });

    it('应返回 401 当未登录', async () => {
      await request(app.getHttpServer())
        .get('/api/v1/rooms')
        .expect(401);
    });
  });

  describe('POST /rooms', () => {
    it('应成功创建房间（管理者）', async () => {
      const createDto = {
        floor: 1,
        roomNumber: '101',
        roomType: 'SINGLE',
        area: 20,
      };

      const response = await request(app.getHttpServer())
        .post('/api/v1/rooms')
        .set('Authorization', `Bearer ${adminToken}`)
        .send(createDto)
        .expect(201);

      expect(response.body.success).toBe(true);
      expect(response.body.data.roomNumber).toBe('101');
      expect(response.body.data.status).toBe('VACANT');
      expect(response.body.data.cleanStatus).toBe('CLEAN');

      // 验证数据库中是否创建成功
      const room = await prisma.room.findUnique({
        where: { id: response.body.data.id },
      });
      expect(room).toBeDefined();
    });

    it('应返回 403 当租客尝试创建房间', async () => {
      const createDto = {
        floor: 1,
        roomNumber: '102',
        roomType: 'SINGLE',
        area: 20,
      };

      await request(app.getHttpServer())
        .post('/api/v1/rooms')
        .set('Authorization', `Bearer ${tenantToken}`)
        .send(createDto)
        .expect(403);
    });

    it('应返回 400 当房间号重复', async () => {
      const createDto = {
        floor: 1,
        roomNumber: '101',  // 已存在
        roomType: 'SINGLE',
        area: 20,
      };

      const response = await request(app.getHttpServer())
        .post('/api/v1/rooms')
        .set('Authorization', `Bearer ${adminToken}`)
        .send(createDto)
        .expect(400);

      expect(response.body.error.code).toBe('ROOM_ALREADY_EXISTS');
    });
  });

  describe('PATCH /rooms/:id/clean', () => {
    it('应成功标记房间为已清洁（保洁）', async () => {
      // 创建待清洁房间
      const room = await prisma.room.create({
        data: {
          buildingId: 'default-building',
          floor: 2,
          roomNumber: '201',
          roomType: 'SINGLE',
          status: 'VACANT',
          cleanStatus: 'PENDING_CLEAN',
        },
      });

      const cleanerToken = await createTestCleaner(prisma);

      const response = await request(app.getHttpServer())
        .patch(`/api/v1/rooms/${room.id}/clean`)
        .set('Authorization', `Bearer ${cleanerToken}`)
        .expect(200);

      expect(response.body.data.cleanStatus).toBe('CLEAN');

      // 验证数据库更新
      const updatedRoom = await prisma.room.findUnique({
        where: { id: room.id },
      });
      expect(updatedRoom.cleanStatus).toBe('CLEAN');
    });
  });
});
```

## 4. 核心业务测试场景清单

### 4.1 房源管理

| 场景 | 操作步骤 | 预期结果 |
|------|----------|----------|
| **新建房间** | 管理者创建房间，输入楼层、房号、房型 | 房间创建成功，状态为 VACANT + CLEAN |
| **房间重复** | 创建已存在的房间号 | 返回错误"房间号已存在" |
| **标记已清洁** | 保洁标记待清洁房间为已清洁 | 清洁状态变为 CLEAN |
| **保洁权限** | 保洁尝试修改房间信息 | 返回 403 权限不足 |
| **租客查看** | 租客查看房间列表 | 只返回自己租住的房间 |

### 4.2 合同管理

| 场景 | 操作步骤 | 预期结果 |
|------|----------|----------|
| **创建合同** | 管理者创建合同，选择房间和租客 | 合同创建成功，生成邀请码，房间状态变为 RESERVED |
| **房间已占用** | 创建合同时选择已被占用的房间 | 返回错误"房间已被占用" |
| **房间未清洁** | 创建合同时选择未清洁的房间 | 返回错误"房间未清洁，无法出租" |
| **合同生效** | 合同生效日期到达 | 合同状态自动变为 ACTIVE，房间状态变为 OCCUPIED |
| **合同到期** | 合同到期日期到达 | 合同状态自动变为 EXPIRED，发送到期通知 |
| **提前退租** | 管理者终止合同 | 合同状态变为 TERMINATED，房间状态变为 VACANT + PENDING_CLEAN |
| **续签合同** | 管理者续签合同 | 旧合同状态变为 RENEWED，创建新合同并关联旧合同 |

### 4.3 账单与收租

| 场景 | 操作步骤 | 预期结果 |
|------|----------|----------|
| **生成账单** | 定时任务生成月租金账单 | 账单创建成功，状态为 PENDING，快照保存租金金额 |
| **部分收款** | 管理者确认部分收款 | 账单状态变为 PARTIAL，已缴金额更新 |
| **全额收款** | 管理者确认全额收款 | 账单状态变为 PAID，创建收款记录 |
| **账单逾期** | 到期日过后账单未缴清 | 账单状态自动变为 OVERDUE，计算滞纳金，发送提醒 |
| **滞纳金计算** | 逾期10天，宽限期3天，日比例0.003 | 滞纳金 = 租金 × 0.003 × 7天 |
| **滞纳金封顶** | 滞纳金超过封顶金额 | 滞纳金 = 封顶金额 |
| **收款幂等** | 使用相同幂等键重复收款 | 返回第一次收款的结果，不重复扣款 |

### 4.4 预付账户与水电

| 场景 | 操作步骤 | 预期结果 |
|------|----------|----------|
| **创建账户** | 租客签订合同后自动创建预付账户 | 账户创建成功，余额为 0 |
| **充值** | 管理者为租客充值 | 创建充值流水，余额增加 |
| **录入读数** | 管理者录入水电读数 | 计算用量，自动扣费，创建扣费流水，余额减少 |
| **余额不足** | 录入读数时余额不足以扣费 | 扣费失败，发送低余额提醒 |
| **余额查询** | 租客查看余额 | 显示当前余额（从流水汇总计算） |
| **流水记录** | 租客查看流水 | 显示所有充值、扣费、退款记录 |
| **单价变更** | 管理者修改水电单价 | 单价生效日期之后的读数使用新单价 |

### 4.5 报修工单

| 场景 | 操作步骤 | 预期结果 |
|------|----------|----------|
| **租客报修** | 租客提交报修，填写问题和上传照片 | 工单创建成功，状态为 SUBMITTED，管理者收到通知 |
| **管理者受理** | 管理者受理工单 | 工单状态变为 ACKNOWLEDGED，租客收到通知 |
| **派单** | 管理者填写维修方并派单 | 工单状态变为 ASSIGNED，记录派单时间和维修方 |
| **开始维修** | 管理者标记"维修中" | 工单状态变为 IN_PROGRESS |
| **完成维修** | 管理者标记"待确认" | 工单状态变为 PENDING_CONFIRM，通知租客确认 |
| **租客确认** | 租客确认完成 | 工单状态变为 COMPLETED，记录完成时间 |
| **租客反馈问题** | 租客反馈问题未解决 | 工单状态回退到 IN_PROGRESS，通知管理者 |
| **自动确认** | 3天内租客未反馈 | 工单状态自动变为 COMPLETED |
| **取消工单** | 租客取消刚提交的工单 | 工单状态变为 CANCELLED（仅限 SUBMITTED 状态） |

### 4.6 退租结算

| 场景 | 操作步骤 | 预期结果 |
|------|----------|----------|
| **发起退租** | 管理者发起退租，选择退租日期 | 创建退租记录（DRAFT），生成结算单（建议值） |
| **租金折算** | 系统计算未住满天数的租金退款 | 按"实际天数÷当月天数"计算退款金额 |
| **押金退还** | 系统计算押金退还金额 | 默认全额退还，管理者可调整扣除项 |
| **水电余额** | 系统查询预付账户余额 | 余额退还金额添加到结算单 |
| **欠费扣除** | 租客有未缴账单 | 从押金中扣除欠费金额 |
| **调整结算单** | 管理者修改结算单明细 | 调整成功，记录调整原因 |
| **提交结算单** | 管理者提交结算单 | 状态变为 PENDING，结算单锁定不可修改 |
| **确认退款** | 管理者确认退款 | 状态变为 COMPLETED，合同变 TERMINATED，房间变 PENDING_CLEAN，预付账户余额退还 |

### 4.7 权限控制

| 场景 | 操作步骤 | 预期结果 |
|------|----------|----------|
| **租客访问他人账单** | 租客尝试访问其他租客的账单 | 返回 403 或 404，不泄露其他租客信息 |
| **保洁查看租客信息** | 保洁尝试查看租客列表 | 返回 403 权限不足 |
| **租客修改房间状态** | 租客尝试修改房间状态 | 返回 403 权限不足 |
| **未登录访问** | 未登录用户访问需鉴权的接口 | 返回 401 未登录 |
| **Token 过期** | 使用过期 Token 访问接口 | 返回 401 Token 已过期 |

### 4.8 邀请码机制

| 场景 | 操作步骤 | 预期结果 |
|------|----------|----------|
| **生成邀请码** | 管理者创建合同 | 自动生成邀请码，48小时有效 |
| **租客注册** | 租客使用邀请码注册 | 注册成功，账号绑定到合同和租客，邀请码失效 |
| **邀请码无效** | 租客使用不存在的邀请码 | 返回错误"邀请码无效" |
| **邀请码过期** | 租客使用过期邀请码 | 返回错误"邀请码已过期" |
| **邀请码重复使用** | 租客使用已使用的邀请码 | 返回错误"邀请码已被使用" |

| 场景 | 测试步骤 | 预期结果 |
|------|----------|----------|
| **SQL 注入** | 输入 `' OR '1'='1` 到查询字段 | 查询失败，不返回非法数据 |
| **XSS 攻击** | 输入 `<script>alert(1)</script>` 到文本字段 | 内容被转义，不执行脚本 |
| **权限越权** | 租客 A 尝试访问租客 B 的账单 | 返回 403 或 404 |
| **暴力破解** | 连续尝试登录10次以上 | 触发限流，返回 429 |
| **Token 伪造** | 使用伪造的 JWT Token | 返回 401 Token 无效 |
